import numpy as np

diem_5_mon = np.array([8.5, 7.5, 9.0, 8.0, 8.5])
print("Kích thước vector:", diem_5_mon.shape) # (5,)

tong_diem = np.sum(diem_5_mon)
diem_tb   = np.mean(diem_5_mon)
diem_max  = np.max(diem_5_mon)

print("Tổng điểm 5 môn (∑):", tong_diem)
print("Điểm trung bình (x̄):", round(diem_tb, 2))
print("Môn điểm cao nhất  :", diem_max)
if diem_tb >= 8.0:
    print("🎉 Chúc mừng: Đạt Học bổng Khuyến khích!")

import numpy as np

anh_toi = np.array([
    [10,  20,  30,  15],
    [40, 110, 120,  35],
    [45, 130, 100,  25],
    [10,  15,  20,  10]
])

print("Độ sáng TB ban đầu:", np.mean(anh_toi)) # 45.9375

anh_sang = np.clip(anh_toi * 1.5 + 60, 0, 255)
print("Ma trận ảnh sau khi cứu sáng:\n", anh_sang)
print("Độ sáng TB mới:", np.mean(anh_sang))

anh_nhi_phan = np.where(anh_sang >= 128, 255, 0)
print("Ảnh nhị phân tách nền:\n", anh_nhi_phan)
